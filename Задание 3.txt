#!/bin/bash

# Проверка наличия входного файла
if [ $# -ne 1 ]; then
  echo "Использование: $0 <input_file.gff>"
  exit 1
fi

input_file=$1

# Проверка существования входного файла
if [ ! -f "$input_file" ]; then
  echo "Ошибка: Файл $input_file не найден."
  exit 1
fi

# Создание директории
mkdir -p terminal_task/results

# Шаг 1: Фильтрация строк по третьей колонке (gene) и наличию "gene_type=unprocessed_pseudogene" в 9 колонке
# Шаг 2: Оставляем только столбцы 1,4,5,7,9
# Шаг 3: Извлекаем из 9 колонки только gene_name
grep -P "\tgene\t" "$input_file" | grep "gene_type=unprocessed_pseudogene" \
| cut -f1,4,5,7,9 \
| sed -E 's/.*gene_name=([^;]+);?.*/\1/' > terminal_task/results/temp.tsv

# Шаг 4: Модификация координат в зависимости от цепи
awk -F'\t' '{
  if ($4 == "+") {
    start = $2
    end = start + 1
  } else if ($4 == "-") {
    start = $3
    end = start + 1
  }
  print $1 "\t" start "\t" end "\t" $5
}' terminal_task/results/temp.tsv > terminal_task/results/result.tsv

# Удаляем временный файл
rm terminal_task/results/temp.tsv

echo "Результат записан в terminal_task/results/result.tsv"
