#!/bin/bash

#Проверка количества аргументов
if [ "$" -ne2 ]; then
  echo "Ошибка: необходимо передать два аргумента (-u или -i и слово)."
  exit 1
fi

option=$1
word=$2

#Обработка аргументов
case "$option" in
  -u) echo "${word^^}";;
#Преобразование в верхний регистр
  -l) echo "${word,,}";;
#Преобразование в нижний регистр
  *) echo "Ошибка: неверный аргумент. Используйте -u для верхнего регистра или -l для нижнего регистра."
;;
esac
