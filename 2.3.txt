#!/bin/bash

# Проверяем количество аргументов
if [ "$#" -ne 3 ]; then
  echo "Ошибка: необходимо передать ровно 3 аргумента."
  exit 1
else
#Вывод аргументов при успешной проверке
  echo "Начинаем обработку..."
  echo "Аргумент 1: $1"
  echo "Аргумент 2: $2"
  echo "Аргумент 3: $3"
fi
